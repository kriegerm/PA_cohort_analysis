#### ANVIO COMMANDS#####
# Below is documentation of the steps to download the genomes from NCBI and run the analysis. Genomes can be found listed in Table S07.


# Download genomes
conda activate ncbi_databsets
datasets download genome accession \
  --inputfile accessions.txt \
  --include genome \
  --assembly-source refseq \
  --dehydrated

unzip ncbi_dataset.zip -d genomes
datasets rehydrate --directory genomes
cd genomes
mv ncbi_dataset/data/*/*.fna ./
mv ncbi_dataset/data/*/*.fna ./genomes


#Change the file names
for item in *.fna;
do 
	storevar=$(echo `ls $item | cut -d"." -f1`)
	mv $item ${storevar}.fasta
done

ls *fasta | awk 'BEGIN{FS=".fasta"}{print $1}' > genomes.txt

conda deactivate
conda activate anvio-8

# ----- Put the below code in a bash script to run ----- #

for g in `cat genomes.txt`
do
    echo
    echo "Working on $g ..."
    echo
    anvi-script-reformat-fasta ${g}.fasta \
                               --min-len 2500 \
                               --simplify-names \
                               -o ${g}_scaffolds_2.5K.fasta
done


for g in `cat genomes.txt`
do
    echo
    echo "Working on $g ..."
    echo
    anvi-gen-contigs-database -f ${g}_scaffolds_2.5K.fasta \
                              -o ${g}.db \
                              --num-threads 4 \
                              -n ${g}
done


for g in *.db
do
    anvi-run-hmms -c $g --num-threads 4
    anvi-run-ncbi-cogs -c $g --num-threads 4
    anvi-scan-trnas -c $g --num-threads 4
    anvi-run-scg-taxonomy -c $g --num-threads 4
    anvi-run-kegg-kofams -c $g --num-threads 4

done


# ----- End bash script ----- #

mkdir dbs
cp genomes/*db dbs
cd dbs 

#LOOK FOR CONTAMINATION
anvi-script-gen-genomes-file --input-dir ./ -o external-genomes.txt
anvi-estimate-genome-completeness -e external-genomes.txt


##LOOK AT GENE STATS
anvi-display-contigs-stats *db

anvi-pan-genome -g PA_GENOMES.db --project-name PA_PANGENOME --num-threads 4 2>> pangenome_log.txt
